import { useFrame, useLoader, useThree } from '@react-three/fiber';
import * as THREE from 'three';
import { lessons } from '../lessons';
import type { Lesson, ShowcaseState, Station } from '../teaching/types';
import {
  ConstructionModel,
  RiverModel,
  SpatialBoard,
} from './spatial-teaching';
import { Interior, ImportedBook } from './asset-interior';

import type { ScenePresentation } from './storyboard';
const ANCHORS: Record<Station, V3> = {
  desk: [-1.65, 0.568, -0.9],
  bookshelf: [-2.15, 0.568, -3.1],
  board: [-2.1, 0.568, -2.8],
  experiment: [0.45, 0.568, -0.1],
};
const PICKS: Record<string, V3> = {
  desk: [-3.1, 0.883, 0.95],
  bookshelf: [-3.65, 1.7, -3.4],
  board: [0.8, 1.7, -3.6],
  experiment: [2.3, 0.824, -0.1],
};
for (const [id, x, y] of [
  ['root', 800, 305],
  ['branch-0', 320, 490],
  ['branch-1', 800, 490],
  ['branch-2', 1280, 490],
  ['leaf-0-0', 222, 700],
  ['leaf-1-0', 702, 700],
] as const) {
  PICKS[id] = [
    BOARD_POSITION[0] + (x / 1600 - 0.5) * BOARD_SIZE[0],
    BOARD_POSITION[1] + (0.5 - y / 900) * BOARD_SIZE[1],
    BOARD_POSITION[2] + 0.01,
  ];
}
interface WorldMetrics {
  transitions: { preset: string; frames: number; ms: number; fps: number }[];
  renderer: string;
  triangles: number;
  drawCalls: number;
  picks: Record<string, { x: number; y: number }>;
  stationClicks: string[];
  books: Record<string, { x: number; y: number; z: number }>;
  bookAnimation?: { id: string; completed: boolean; durationMs: number };
  ready: boolean;
  budget?: ReturnType<typeof assetBudget>;
  camera?: { position: number[]; quaternion: number[]; zoom: number };
}
declare global {
  interface Window {
    __PETROS_WORLD_METRICS__?: WorldMetrics;
  }
}
export interface SceneProps {
  state: ShowcaseState;
  caption?: string;
  progress?: number;
  scene: ScenePresentation;
  reduced: boolean;
  frozen: boolean;
  previewId?: string;
  onReady: () => void;
  onBookArrived: () => void;
  onStation: (s: Station) => void;
  onBook: (l: Lesson) => void;
  onPick: Pick;
  onAnchor: (point: AvatarProjection) => void;
  onCamera: (busy: boolean) => void;
}
function bookStart(lesson: Lesson): V3 {
  if (lesson.general) return [lesson.id === 'time' ? -3.4 : -3.0, 1.08, 0.95];
  return [
    lesson.id === 'ai3' || lesson.id === 'ehs3' ? -3.5 : -3.2,
    lesson.id.startsWith('ai') ? 2.23 : 1.575,
    -3.54,
  ];
}
for (const lesson of lessons) PICKS[`book-${lesson.id}`] = bookStart(lesson);
function SourceBook({
  lesson,
  active,
  phase,
  paused,
  reduced,
  onBook,
  onArrived,
}: {
  lesson: Lesson;
  active: boolean;
  phase: ScenePresentation['phase'];
  paused: boolean;
  reduced: boolean;
  onBook: (l: Lesson) => void;
  onArrived: () => void;
}) {
  const held = useRef(false);
  const group = useRef<THREE.Group>(null);
  const elapsed = useRef(0);
  const started = useRef(0);
  const done = useRef(false);
  const { invalidate } = useThree();
  const start = bookStart(lesson);
  const destination: V3 = [-3.15, 0.925, -1.7];
  const moving =
    (active &&
      (phase === 'retrieving' ||
        phase === 'book' ||
        phase === 'teaching' ||
        phase === 'practice' ||
        phase === 'recap')) ||
    (held.current && (phase === 'arrival' || phase === 'library'));
  useEffect(() => {
    if (phase === 'retrieving' && !active) held.current = false;
    if (phase === 'retrieving' && active) {
      elapsed.current = 0;
      started.current = performance.now();
      done.current = false;
      invalidate();
    }
  }, [phase, active, invalidate]);
  useEffect(() => {
    invalidate();
  }, [paused, reduced, invalidate]);
  useFrame((_, delta) => {
    if (!group.current) return;
    if (moving) {
      const p =
        phase === 'retrieving'
          ? reduced
            ? 1
            : Math.min(1, elapsed.current / 0.9)
          : 1;
      if (!paused && phase === 'retrieving')
        elapsed.current += Math.min(delta, 0.06);
      const eased = p * p * (3 - 2 * p);
      group.current.position.set(
        THREE.MathUtils.lerp(start[0], destination[0], eased),
        THREE.MathUtils.lerp(start[1], destination[1], eased) +
          Math.sin(Math.PI * p) * 0.65,
        THREE.MathUtils.lerp(start[2], destination[2], eased),
      );
      group.current.rotation.set((-Math.PI / 2) * eased, 0, 0.1 * eased);
      if (window.__PETROS_WORLD_METRICS__)
        window.__PETROS_WORLD_METRICS__.books[lesson.id] = {
          x: group.current.position.x,
          y: group.current.position.y,
          z: group.current.position.z,
        };
      if (p < 1 && !paused) invalidate();
      else if (p >= 1 && phase === 'retrieving' && !done.current) {
        done.current = true;
        held.current = true;
        if (window.__PETROS_WORLD_METRICS__)
          window.__PETROS_WORLD_METRICS__.bookAnimation = {
            id: lesson.id,
            completed: true,
            durationMs: reduced ? 0 : performance.now() - started.current,
          };
        onArrived();
      }
    } else {
      group.current.position.set(...start);
      group.current.rotation.set(0, 0, 0);
    }
  });
  return (
    <group
      ref={group}
      position={start}
      scale={0.37}
      onClick={(e) => {
        e.stopPropagation();
        onBook(lesson);
      }}
      onPointerOver={() => (document.body.style.cursor = 'pointer')}
      onPointerOut={() => (document.body.style.cursor = '')}
    >
      <group rotation={[0, -Math.PI / 2, 0]}>
        <ImportedBook
          variant={lesson.module ?? (lesson.id === 'time' ? 15 : 2)}
          height={1.04}
        />
      </group>
      <Label
        text={
          lesson.general ? 'EXPLORE' : lesson.id.startsWith('ai') ? 'AI' : 'EHS'
        }
        position={[0, 0.24, 0.105]}
        size={[0.61, 0.15]}
        color="#f8e7be"
        fontSize={115}
      />
      <Label
        text={
          lesson.general
            ? lesson.id === 'time'
              ? 'TIME'
              : 'AI'
            : `Module ${lesson.module}`
        }
        position={[0, -0.06, 0.105]}
        size={[0.61, 0.2]}
        color="#f8e7be"
        fontSize={115}
      />
      {!lesson.general && (
        <Label
          text={
            {
              ai3: 'MINIMAX',
              ai4: 'REASONING',
              ehs3: 'SITE SAFETY',
              ehs5: 'WATER',
            }[lesson.id as 'ai3' | 'ai4' | 'ehs3' | 'ehs5']
          }
          position={[0, -0.3, 0.105]}
          size={[0.61, 0.12]}
          color="#e1cca2"
          fontSize={80}
        />
      )}
      {active && (
        <Label
          text="●"
          position={[0, 0.66, 0.02]}
          size={[0.22, 0.2]}
          color="#c9d7a7"
        />
      )}
    </group>
  );
}
function Character({
  station,
  state,
  scene,
  reduced,
  frozen,
  onAnchor,
}: {
  station: Station;
  state: ShowcaseState;
  scene: ScenePresentation;
  reduced: boolean;
  frozen: boolean;
  onAnchor: SceneProps['onAnchor'];
}) {
  const textures = useLoader(THREE.TextureLoader, [
    '/petros/states/petros-idle.png',
    '/petros/states/petros-listening.png',
    '/petros/states/petros-thinking.png',
    '/petros/states/petros-point-right.png',
  ]);
  const [anchor, setAnchor] = useState<Station>('desk');
  const material = useRef<THREE.SpriteMaterial>(null);
  const elapsed = useRef(1);
  const { invalidate, camera, size } = useThree();
  const first = useRef(true);
  const target = useRef(station);
  const anchorCallback = useRef(onAnchor);
  anchorCallback.current = onAnchor;
  useEffect(() => {
    target.current = station;
    if (first.current) {
      first.current = false;
      setAnchor(station);
      elapsed.current = 0.8;
    } else elapsed.current = 0;
    invalidate();
  }, [station, invalidate]);
  useEffect(() => {
    invalidate();
  }, [frozen, state.paused, invalidate]);
  useFrame((_, delta) => {
    if (!material.current) return;
    if (elapsed.current < 0.8) {
      if (!frozen && !state.paused) elapsed.current += delta;
      const t = reduced ? 0.8 : elapsed.current;
      material.current.opacity =
        t < 0.15 ? 1 - t / 0.15 : t < 0.65 ? 0 : Math.min(1, (t - 0.65) / 0.15);
      if (t >= 0.15 && anchor !== target.current) setAnchor(target.current);
      if (t < 0.8 && !frozen && !state.paused) invalidate();
    }
    const rect = document
      .querySelector('[data-world-scene] canvas')
      ?.getBoundingClientRect();
    const project = (dy: number) => {
      const p = new THREE.Vector3(...ANCHORS[anchor]);
      p.y += dy;
      p.project(camera);
      return {
        x: (rect?.left ?? 0) + ((p.x + 1) * size.width) / 2,
        y: (rect?.top ?? 0) + ((1 - p.y) * size.height) / 2,
      };
    };
    anchorCallback.current({
      head: project(0.7),
      feet: project(-0.55),
      visible: material.current.opacity > 0.25,
    });
  });
  const pose =
    frozen || state.pendingRequest
      ? 1
      : state.cue?.kind === 'thought'
        ? 2
        : scene.phase === 'teaching' &&
            station === 'board' &&
            state.highlightTargetId
          ? 3
          : 0;
  return (
    <group>
      <sprite position={ANCHORS[anchor]} scale={[1.15, 1.54, 1]}>
        <spriteMaterial
          ref={material}
          map={textures[pose]}
          transparent
          alphaTest={0.03}
          depthWrite={false}
          toneMapped={false}
        />
      </sprite>
      <mesh
        rotation={[-Math.PI / 2, 0, 0]}
        position={[ANCHORS[anchor][0], 0.012, ANCHORS[anchor][2]]}
        scale={[0.5, 0.18, 1]}
      >
        <circleGeometry args={[0.5, 24]} />
        <meshBasicMaterial
          color="#29221c"
          transparent
          opacity={0.22}
          depthWrite={false}
        />
      </mesh>
    </group>
  );
}
function CameraRig({
  scene,
  state,
  reduced,
  frozen,
  onCamera,
}: {
  scene: ScenePresentation;
  state: ShowcaseState;
  reduced: boolean;
  frozen: boolean;
  onCamera: SceneProps['onCamera'];
}) {
  const { camera, size, invalidate, gl, scene: threeScene } = useThree();
  const look = useRef(new THREE.Vector3(-0.4, 1.5, -3));
  const initialized = useRef(false),
    lastSkip = useRef(scene.skipCamera);
  const freeze = useRef(frozen);
  freeze.current = frozen;
  const transition = useRef<{
    t: number;
    wall: number;
    position: THREE.Vector3;
    look: THREE.Vector3;
    fov: number;
    frames: number;
    duration: number;
  } | null>(null);
  const library = ['library', 'retrieving', 'book'].includes(scene.phase);
  const experiment = state.sceneStation === 'experiment';
  useEffect(() => {
    if (freeze.current && initialized.current) return;
    const instant =
      !initialized.current || lastSkip.current !== scene.skipCamera;
    initialized.current = true;
    lastSkip.current = scene.skipCamera;
    transition.current = {
      t: 0,
      wall: performance.now(),
      position: camera.position.clone(),
      look: look.current.clone(),
      fov: (camera as THREE.PerspectiveCamera).fov,
      frames: 0,
      duration: reduced || instant ? 0 : 0.9,
    };
    onCamera(true);
    invalidate();
  }, [
    scene.preset,
    library,
    experiment,
    scene.skipCamera,
    size.width,
    size.height,
    reduced,
  ]);
  useFrame((_, delta) => {
    const t = transition.current;
    if (!t) return;
    if (freeze.current && t.duration > 0) {
      transition.current = null;
      onCamera(false);
      return;
    }
    const focus = scene.preset === 'teaching' || scene.preset === 'interaction';
    const destination = library
      ? new THREE.Vector3(-2.8, 1.6, 0.1)
      : focus && experiment
        ? new THREE.Vector3(2.5, 1.78, 2.6)
        : focus
          ? new THREE.Vector3(-0.55, 1.65, 1.85)
          : new THREE.Vector3(0.2, 1.65, 3.95);
    const target = library
      ? new THREE.Vector3(-2.8, 1.27, -3.55)
      : focus && experiment
        ? new THREE.Vector3(1.8, 1.24, -0.2)
        : focus
          ? new THREE.Vector3(size.width < 600 ? 0.8 : 0.0, 1.3, -3.6)
          : new THREE.Vector3(-0.1, 1.35, -2.9);
    const distance = destination.distanceTo(target);
    const span =
      focus && !experiment
        ? { w: size.width < 600 ? 4.7 : 6.5, h: 3.35 }
        : library
          ? { w: 4.0, h: 3.35 }
          : focus
            ? { w: 3.7, h: 2.25 }
            : null;
    const fov = span
      ? THREE.MathUtils.radToDeg(
          2 *
            Math.atan(
              Math.max(span.h / 2, span.w / 2 / (size.width / size.height)) /
                distance,
            ),
        )
      : 60;
    t.t += Math.min(delta, 0.06);
    const p = t.duration ? Math.min(1, t.t / t.duration) : 1,
      e = p * p * (3 - 2 * p);
    camera.position.lerpVectors(t.position, destination, e);
    look.current.lerpVectors(t.look, target, e);
    camera.lookAt(look.current);
    (camera as THREE.PerspectiveCamera).fov = THREE.MathUtils.lerp(
      t.fov,
      fov,
      e,
    );
    camera.updateProjectionMatrix();
    t.frames++;
    const snapshot = {
      position: camera.position.toArray(),
      quaternion: camera.quaternion.toArray(),
      zoom: 1,
      fov: (camera as THREE.PerspectiveCamera).fov,
    };
    if (window.__PETROS_WORLD_METRICS__)
      window.__PETROS_WORLD_METRICS__.camera = snapshot;
    if (p < 1) invalidate();
    else {
      const previous = window.__PETROS_WORLD_METRICS__;
      const picks: WorldMetrics['picks'] = {};
      for (const [id, point] of Object.entries(PICKS)) {
        const v = new THREE.Vector3(...point).project(camera);
        picks[id] = {
          x: ((v.x + 1) * size.width) / 2,
          y: ((1 - v.y) * size.height) / 2,
        };
      }
      let triangles = 0;
      threeScene.traverse((o) => {
        if (o instanceof THREE.Mesh) {
          const g = o.geometry;
          triangles +=
            ((g.index ? g.index.count : (g.attributes.position?.count ?? 0)) /
              3) *
            (o instanceof THREE.InstancedMesh ? o.count : 1);
        }
      });
      window.__PETROS_WORLD_METRICS__ = {
        transitions: [
          ...(previous?.transitions ?? []),
          {
            preset: library
              ? 'library'
              : focus && experiment
                ? 'experiment'
                : scene.preset,
            frames: t.frames,
            ms: performance.now() - t.wall,
            fps: t.duration
              ? (t.frames * 1000) / (performance.now() - t.wall)
              : 0,
          },
        ].slice(-30),
        budget: assetBudget(threeScene, gl),
        renderer: gl.getContext().getParameter(gl.getContext().RENDERER),
        triangles,
        drawCalls: gl.info.render.calls,
        picks,
        stationClicks: previous?.stationClicks ?? [],
        books: previous?.books ?? {},
        bookAnimation: previous?.bookAnimation,
        ready: true,
        camera: snapshot,
      };
      transition.current = null;
      onCamera(false);
    }
  });
  return null;
}
function SpatialBounds() {
  const { camera, size } = useThree();
  useFrame(() => {
    const points = [];
    for (const x of [1.22, 3.4])
      for (const y of [0.8, 2.17])
        for (const z of [-0.9, 0.65]) {
          const v = new THREE.Vector3(x, y, z).project(camera);
          points.push({
            x: ((v.x + 1) * size.width) / 2,
            y: ((1 - v.y) * size.height) / 2,
          });
        }
    const xs = points.map((p) => p.x),
      ys = points.map((p) => p.y);
    (window as any).__PETROS_EXPERIMENT_RECT__ = {
      x: Math.min(...xs),
      y: Math.min(...ys),
      w: Math.max(...xs) - Math.min(...xs),
      h: Math.max(...ys) - Math.min(...ys),
    };
  });
  return null;
}
function Ready({ onReady }: { onReady: () => void }) {
  const fired = useRef(false);
  useFrame(() => {
    if (!fired.current) {
      fired.current = true;
      onReady();
    }
  });
  return null;
}
export function Scene(props: SceneProps) {
  const {
    state,
    scene,
    reduced,
    frozen,
    previewId,
    caption,
    progress,
    onReady,
    onBookArrived,
    onBook,
    onPick,
    onAnchor,
    onCamera,
  } = props;
  useEffect(
    () => () => {
      document.body.style.cursor = '';
    },
    [],
  );
  return (
    <>
      <Interior onStation={props.onStation} />
      {lessons.map((l) => (
        <SourceBook
          key={l.id}
          lesson={l}
          active={l.id === (previewId ?? state.lessonId)}
          phase={scene.phase}
          paused={state.paused || frozen}
          reduced={reduced}
          onBook={onBook}
          onArrived={onBookArrived}
        />
      ))}
      <SpatialBoard
        state={state}
        reveal={scene.reveal}
        onPick={onPick}
        visible={['teaching', 'practice', 'recap'].includes(scene.phase)}
        caption={caption}
        progress={progress}
        onOpen={() => props.onStation('board')}
      />
      <group position={[0.35, -0.025, -0.59]} scale={0.65}>
        {state.visualState.kind === 'hazards' ? (
          <ConstructionModel
            reduced={reduced}
            state={state}
            angle={state.visualState.angle ?? 0}
            onPick={onPick}
          />
        ) : state.visualState.kind === 'river' ? (
          <RiverModel
            state={state}
            angle={state.visualState.angle ?? 0}
            onPick={onPick}
          />
        ) : null}
      </group>
      <SpatialBounds />
      <Character
        station={scene.phase === 'arrival' ? 'desk' : state.sceneStation}
        state={state}
        scene={scene}
        reduced={reduced}
        frozen={frozen}
        onAnchor={onAnchor}
      />
      <CameraRig
        scene={scene}
        state={state}
        reduced={reduced}
        frozen={frozen || state.paused}
        onCamera={onCamera}
      />
      <Ready onReady={onReady} />
    </>
  );
}
